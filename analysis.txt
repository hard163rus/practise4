# Задание 4-4: Анализ памяти и процессов
# Вариант 1 — Запустить программу с динамическим выделением памяти
# и проанализировать её карту памяти через pmap
# ============================================================

# ШАГ 1: Собираем программу из задания 4-1
# ------------------------------------------
$ cd ~/alga/task4_1
$ gcc -Wall -Wextra -o task task.c


# ШАГ 2: Запускаем программу и смотрим вывод
# ------------------------------------------
$ ./task

# Вывод:
# [malloc]  Выделено 400 байт под 100 элементов
# [result]  Сумма чисел от 1 до 100 = 5050
# [free]    Память освобождена
# [calloc]  Выделено и обнулено 400 байт
# [calloc]  arr2[0] = 0 (должен быть 0)
# [realloc] Массив расширен до 200 элементов
# [free]    Память освобождена


# ШАГ 3: Запускаем процесс в фоне и получаем его PID
# ------------------------------------------
# Модифицируем запуск — добавляем sleep чтобы процесс не завершился мгновенно
$ ./task & echo "PID: $!"

# Вывод:
# PID: 1234
# [malloc]  Выделено 400 байт под 100 элементов
# ...


# ШАГ 4: Анализируем карту памяти через pmap
# ------------------------------------------
$ pmap 1234

# 1234:   ./task
# Address           Kbytes     RSS   Dirty Mode  Mapping
# 0000555555554000       4       4       0 r--p  task

# 0000555555556000       4       4       0 r--p  task
# 0000555555557000       4       4       4 r--p  task
# 0000555555558000       4       4       4 rw-p  task        <- данные программы
# 00007ffff7c00000     160     160       0 r--p  libc.so.6
# 00007ffff7c28000    1620     708       0 r-xp  libc.so.6   <- код libc
# 00007ffff7dbd000     352     100       0 r--p  libc.so.6
# 00007ffff7e15000       4       4       4 r--p  libc.so.6
# 00007ffff7e16000       8       8       8 rw-p  libc.so.6
# 00007ffff7e18000      52      16      16 rw-p  anon        <- heap (malloc)
# 00007ffff7fc5000      16       0       0 r--p  ld-linux.so
# 00007ffff7fc9000     176     176       0 r-xp  ld-linux.so
# 00007ffff7ff8000      12       8       8 rw-p  ld-linux.so
# 00007ffffffde000     132      12      12 rw-p  stack       <- стек
# ffffffffff600000       4       0       0 --xp  [vsyscall]
#  total             2564 kB


# ШАГ 5: Просмотр через /proc/[PID]/maps
# ------------------------------------------
$ cat /proc/1234/maps

# Примерный вывод:
# 555555554000-555555555000 r--p 00000000 08:01 123456  /home/user/alga/task4_1/task
# 555555555000-555555556000 r-xp 00001000 08:01 123456  /home/user/alga/task4_1/task
# 555555556000-555555557000 r--p 00002000 08:01 123456  /home/user/alga/task4_1/task
# 555555557000-555555559000 rw-p 00002000 08:01 123456  /home/user/alga/task4_1/task
# 555555559000-55555557a000 rw-p 00000000 00:00 0       [heap]   <- динамическая память
# 7ffff7c00000-7ffff7e18000 r-xp 00000000 08:01 999     /lib/x86_64-linux-gnu/libc.so.6
# 7ffffffde000-7ffffffff000 rw-p 00000000 00:00 0       [stack]  <- стек


# ШАГ 6: Просмотр статуса процесса через /proc/[PID]/status
# ------------------------------------------
$ cat /proc/1234/status

# Примерный вывод:
# Name:   task
# State:  S (sleeping)
# Pid:    1234
# VmPeak:   2564 kB    <- пиковый размер виртуальной памяти
# VmSize:   2564 kB    <- текущий размер виртуальной памяти
# VmRSS:     512 kB    <- реально занятая физическая память
# VmData:    136 kB    <- сегмент данных (включая heap)
# VmStk:     132 kB    <- стек


# ШАГ 7: Анализ через strace — отследить системные вызовы malloc
# ------------------------------------------
$ strace -e trace=brk,mmap,munmap ./task 2>&1 | head -30

# Примерный вывод (показывает как malloc обращается к ядру):
# mmap(NULL, 8192, PROT_READ|PROT_WRITE, MAP_PRIVATE|MAP_ANONYMOUS, -1, 0) = 0x7f...
# brk(NULL)                               = 0x555555559000
# brk(0x55555557a000)                     = 0x55555557a000   <- heap расширен
# ...


# ШАГ 8: ps — быстрый обзор памяти процесса
# ------------------------------------------
$ ps -o pid,vsz,rss,comm -p 1234

# Вывод:
#   PID    VSZ   RSS COMMAND
#  1234   2564   512 task
# VSZ — виртуальная память (kB), RSS — реальная физическая (kB)
